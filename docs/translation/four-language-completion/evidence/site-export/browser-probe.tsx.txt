export default function Page(){
 useEffect(()=>{
  const run=async(language:string,mode:string)=>{
   const source=fixtures.cases[1],facts=structuredClone(source.facts) as unknown as ReportSiteFacts,location=source.location as LocationData;
   const crop=facts.crop!,snapshot=crop.snapshot!;
   snapshot.capturedAt='2026-12-05T08:00:00Z';snapshot.beds[0].label='Ṱanḓa {count} $&';
   snapshot.plantings[0].variety='Original ṋḓḽ';

   const t=(source:string)=>reportDocumentText(source,language);
   const heading=t('Site'),tableHeading=t('Crop');
   const markdown=(mode==='legacy'?'':'# Ṱanḓa saved report\n\n')+`## ${heading}\n${privateRecord}\n\n### ${t('Saved plan')}\n`+Array.from({length:9},()=>privateRecord).join(' ')+'\n\n| '+tableHeading+' | '+t('Detail')+' |\n|---|---|\n'+Array.from({length:48},(_,i)=>`| ${i+1} | ${privateRecord} |`).join('\n')+'\n\n'+privateRecord;

   return{bytes:Array.from(new Uint8Array(await blob.arrayBuffer())),unchangedRecords:snapshot.beds[0].label==='Ṱanḓa {count} $&'&&snapshot.plantings[0].variety==='Original ṋḓḽ',sourceRecord:privateRecord};
